const listElement = document.getElementById("anime-list");

function showAnime(list) {
    listElement.innerHTML = "";

    list.forEach(function(anime) {
        const card = document.createElement("div");
        card.className = "card";
        card.innerHTML = `
            <h2>${anime.title}</h2>
            <p class="meta">⭐ ${anime.rating} | ${anime.year}</p>
            <p>${anime.description}</p>
            <p class="genres">${anime.genres.join(", ")}</p>
        `;
        listElement.appendChild(card);
    });
}
 
showAnime(animeData);

const searchInput = document.getElementById("search");

searchInput.addEventListener("input", function() {
    const keyword = searchInput.value.toLowerCase();

    const result = animeData.filter(function(anime) {
        return anime.title.toLowerCase().includes(keyword);
    });

    showAnime(result);
});

const filterButtons = document.querySelectorAll("#filters button");

filterButtons.forEach(function(button) {
    button.addEventListener("click", function() {
        const genre = button.dataset.genre;

        filterButtons.forEach(function(b) {
            b.classList.remove("active");
        });
        button.classList.add("active");

        if (genre === "all") {
            showAnime(animeData);
        } else {
            const result = animeData.filter(function(anime) {
                return anime.genres.includes(genre);
            });
            showAnime(result);
        }
    });
});
